<script lang="ts">
  // The Reading legend (ADR-0007 / Q2 / Q16): a canvas-anchored radio of the exhibit's competing
  // interpretive passes. Base-only by default (no camp privileged — scholarly-honest); selecting a
  // reading overlays its (colour-coded) notes on the always-visible base. Styled as a canvas overlay
  // to match the Reader's `.popup` (curator's-study system); the active reading's description is the
  // one intent line shown (principle #1), kept compact. Rendered inside Reader's relative container.
  import { readingMarkerStyle } from "@render/core";
  import type { Reading } from "@render/core";

  let { readings, active, onselect, hidden = false, onhiddenchange, count }: {
    readings: Reading[];
    active: string | null;
    onselect: (id: string | null) => void;
    /** Hide-all (declutter): when true, the canvas draws no markers — useful on a dense map to read the
     *  basemap itself. Orthogonal to which reading is active; picking any layer below restores them. */
    hidden?: boolean;
    onhiddenchange?: (hidden: boolean) => void;
    /** Per-layer note count on the CURRENT object (id = null → General/base notes; else a reading's id).
     *  Optional — omitted ⇒ no counts render. The host re-mints it per active object so counts stay live. */
    count?: (id: string | null) => number;
  } = $props();

  // Picking a layer always means "show me markers" — un-hide, then select (the approved restore path).
  const pick = (id: string | null): void => { if (hidden) onhiddenchange?.(false); onselect(id); };

  const activeDesc = $derived(active ? readings.find((r) => r.id === active)?.description : undefined);

  // ── Overflow menu (Archie-84af) ──────────────────────────────────────────────────────────────
  // Replaces the row's horizontal scroll. The constraint that made scrolling right in the first place
  // is unchanged and is the thing to protect: this bar's HEIGHT comes straight out of the image, and
  // wrapping the chips measured 151px of a 698px stage (22%) against 34px (5%) for one row. So the
  // chips still live on exactly one line — the ones that do not fit move into a `+N` popover instead
  // of off the right edge.
  //
  // Measure rather than guess: reading names are author-authored and vary wildly, so a hard-coded
  // chip count would be wrong for every library but the one it was tuned on.
  let rowEl = $state<HTMLDivElement | null>(null);
  let rowW = $state(0);
  /** Natural width of each chip, index-aligned with `chips` below (0 = General notes). */
  let chipW = $state<number[]>([]);
  let moreOpen = $state(false);

  const GAP = 4;       // --space-1, the row's gap
  const MORE_W = 52;   // the `+N` button's own footprint, reserved before deciding what fits

  const chips = $derived([
    { id: null as string | null, name: "General notes", colour: "var(--ink-canvas-muted)" },
    ...readings.map((r) => ({ id: r.id as string | null, name: r.name, colour: r.colour ?? "var(--accent)" })),
  ]);

  /**
   * How many chips fit on one line. Returns `chips.length` before measurement lands, so the first
   * paint renders everything — that is also what MAKES measurement possible, since a chip that is
   * not in the DOM has no width to read.
   */
  const fitCount = $derived.by(() => {
    if (rowW === 0 || chipW.length < chips.length) return chips.length;
    const total = chipW.reduce((a, w) => a + w + GAP, 0) - GAP;
    if (total <= rowW) return chips.length;
    // Something must collapse, so the `+N` button now needs room of its own.
    let used = 0, n = 0;
    for (const w of chipW) {
      const next = used + (n > 0 ? GAP : 0) + w;
      if (next + GAP + MORE_W > rowW) break;
      used = next; n += 1;
    }
    // Never collapse to nothing: one chip plus the menu always beats a bare `+N`.
    return Math.max(1, n);
  });

  const overflow = $derived(chips.slice(fitCount));

  // Re-measure on every resize. Only the chips currently rendered can be read, so widths are CACHED
  // per index — the tail's width is captured on the first (full) paint and stays valid, because a
  // chip's natural width does not depend on how many of its siblings are shown.
  function measure(): void {
    const el = rowEl;
    if (!el) return;
    rowW = el.clientWidth;
    const buttons = el.querySelectorAll<HTMLElement>("[data-chip]");
    const next = [...chipW];
    for (const b of buttons) {
      const i = Number(b.dataset.chip);
      if (Number.isInteger(i)) next[i] = b.offsetWidth;
    }
    if (next.length !== chipW.length || next.some((w, i) => w !== chipW[i])) chipW = next;
  }

  $effect(() => {
    const el = rowEl;
    if (!el) return;
    // Fonts change metrics after first paint, and this bar is measured in ems of a webfont — measuring
    // before they land would cache widths for the fallback face and collapse the wrong chips.
    void document.fonts?.ready.then(measure);
    const ro = new ResizeObserver(() => measure());
    ro.observe(el);
    measure();
    return () => ro.disconnect();
  });

  // The chip list changing (a different object, a different exhibit) invalidates every cached width.
  $effect(() => {
    void chips.length;
    chipW = [];
    moreOpen = false;
  });

  function onDocPointer(e: PointerEvent): void {
    if (!moreOpen) return;
    const t = e.target as Node | null;
    if (t && rowEl?.contains(t)) return;
    moreOpen = false;
  }
</script>

<!-- V47 (Archie-52a0) — the swatch IS the mark, drawn from the same `readingMarkerStyle` call the
     canvas paints with. It used to be a solid disc of the reading's colour, which the canvas never
     draws: a mark is an outline at 0.95 with an 18% fill, so the legend was promising a saturated
     block and the reader was hunting for a faint outline. Rendering it as SVG rather than
     translating the numbers into CSS is the point — fill/fill-opacity/stroke/stroke-opacity/
     stroke-width are handed over verbatim, so there is no second copy of the constants to drift.
     A square, not a circle, for the same reason: the mark vocabulary is rect and polygon. -->
{#snippet swatch(colour: string)}
  <!-- `{@const}` must be the immediate child of a BLOCK — inside the <svg> it is a parse error. -->
  {@const ms = readingMarkerStyle(colour, "normal")}
  <svg class="sw" viewBox="0 0 14 14" aria-hidden="true">
    <rect
      x="1.5" y="1.5" width="11" height="11"
      fill={ms.fill} fill-opacity={ms.fillOpacity}
      stroke={ms.stroke} stroke-opacity={ms.strokeOpacity} stroke-width={ms.strokeWidth}
    />
  </svg>
{/snippet}

<!-- Dismissal for the `+N` popover. Both paths, because either alone strands it: Escape is the
     keyboard contract, and a pointer press elsewhere is what a mouse user will do. Bound at the window
     rather than with a backdrop element so nothing new is laid over the canvas. -->
<svelte:window
  onkeydown={(e) => { if (e.key === "Escape" && moreOpen) { moreOpen = false; } }}
  onpointerdown={onDocPointer}
/>

{#if readings.length > 0}
  <!-- aside = complementary landmark (axe region rule: overlay content must live in a landmark) -->
  <aside class="legend" aria-label="Readings">
    <span class="title">Readings</span>
    <span class="gloss">Compare interpretations</span>
    <!-- The popover lives INSIDE the radiogroup on purpose: `role="radiogroup"` wants its radios as
         descendants, so hoisting the overflow chips into a sibling panel would silently break the
         group for a screen reader even though it looked right. -->
    <div class="opts" class:dimmed={hidden} role="radiogroup" aria-label="Readings of this source" bind:this={rowEl}>
      {#each chips.slice(0, fitCount) as c, i (c.id ?? "__base__")}
        <button
          type="button" role="radio" aria-checked={active === c.id} data-chip={i}
          class="opt" class:on={active === c.id && !hidden}
          style="--rd:{c.colour}" onclick={() => pick(c.id)}
        >
          {@render swatch(c.colour)}<span class="nm">{c.name}</span>{#if count}<span class="ct" title="{count(c.id)} notes on this image">{count(c.id)}</span>{/if}
        </button>
      {/each}
      {#if overflow.length > 0}
        <div class="more-wrap">
          <button
            type="button" class="opt more" aria-haspopup="true" aria-expanded={moreOpen}
            aria-label="{overflow.length} more {overflow.length === 1 ? 'reading' : 'readings'}"
            onclick={() => (moreOpen = !moreOpen)}
          >+{overflow.length}</button>
          {#if moreOpen}
            <!-- Docked to the bar rather than floated over the canvas: Archie-40fe spent a whole
                 ticket getting floating chrome off the image, and a menu that re-covers it would
                 undo that. It hangs BELOW the bar, which is chrome territory already. -->
            <div class="more-menu">
              {#each overflow as c (c.id ?? "__base__")}
                <button
                  type="button" role="radio" aria-checked={active === c.id}
                  class="opt" class:on={active === c.id && !hidden}
                  style="--rd:{c.colour}" onclick={() => { pick(c.id); moreOpen = false; }}
                >
                  {@render swatch(c.colour)}<span class="nm">{c.name}</span>{#if count}<span class="ct" title="{count(c.id)} notes on this image">{count(c.id)}</span>{/if}
                </button>
              {/each}
            </div>
          {/if}
        </div>
      {/if}
    </div>
    {#if activeDesc && !hidden}<p class="desc">{activeDesc}</p>{/if}
    <!-- Hide-all: a declutter toggle, separate from the layer radios (visibility ≠ which reading). -->
    <button type="button" class="hide-toggle" aria-pressed={hidden} onclick={() => onhiddenchange?.(!hidden)}>
      {hidden ? "Show notes" : "Hide all"}
    </button>
  </aside>
{/if}
